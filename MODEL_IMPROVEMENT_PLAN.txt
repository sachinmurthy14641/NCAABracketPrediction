def calculate_matchup_factors(team_a, team_b, team_a_stats, team_b_stats):
    """Calculate style matchup factors"""
    
    matchup_features = {}
    
    # Pace differential
    matchup_features['pace_diff'] = abs(team_a_stats['adj_t'] - team_b_stats['adj_t'])
    matchup_features['fast_vs_slow'] = 1 if (
        (team_a_stats['adj_t'] > 72 and team_b_stats['adj_t'] < 68) or
        (team_a_stats['adj_t'] < 68 and team_b_stats['adj_t'] > 72)
    ) else 0
    
    # Three-point reliance (higher variance)
    matchup_features['high_variance'] = 1 if (
        team_a_stats['efg_o'] > 52 or team_b_stats['efg_o'] > 52
    ) else 0
    
    return matchup_features

class EnsembleModel:
    def __init__(self, kenpom_model, torvik_model):
        self.kenpom_model = kenpom_model
        self.torvik_model = torvik_model
        self.meta_model = LogisticRegression()
    
    def predict_proba(self, X_kenpom, X_torvik):
        # Get predictions from each model
        kenpom_pred = self.kenpom_model.predict_proba(X_kenpom)[:, 1]
        torvik_pred = self.torvik_model.predict_proba(X_torvik)[:, 1]
        
        # Simple average (or use meta_model)
        ensemble_pred = (kenpom_pred + torvik_pred) / 2
        
        return ensemble_pred

from sklearn.calibration import CalibratedClassifierCV

def calibrate_ensemble(ensemble_model, X_val, y_val):
    """Apply isotonic regression to calibrate probabilities"""
    
    calibrated = CalibratedClassifierCV(
        ensemble_model,
        method='isotonic',
        cv='prefit'
    )
    
    calibrated.fit(X_val, y_val)
    
    return calibrated
